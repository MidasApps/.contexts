create-agent-tool.ts — Factory de tool com async generator
import { tool } from 'ai';

export function createAgentTool(config: AgentConfig) {
  return tool({
    description: config.description,
    inputSchema: config.inputSchema,
    async *execute(input, options) {
      // 1. Preliminary: analyzing
      yield { status: 'analyzing', agent: config.id };

      // 2. Build agent-specific prompt
      const system = buildAgentPrompt(config.id, options.experimental_context);

      // 3. Run sub-agent with its own model + tools
      const result = await generateText({
        model: models(config.model),
        system,
        tools: config.tools,
        stopWhen: stepCountIs(config.maxSteps ?? 3),
        prompt: input.query,
        onStepFinish: ({ toolCalls }) => {
          // Yield preliminary results for each internal step
          // (handled via callback → shared state)
        },
      });

      // 4. Final result
      yield { status: 'done', summary: result.text };
      return { text: result.text, toolResults: result.toolResults };
    },
  });
}
orchestrator.ts — O router principal
export function createOrchestrator(context: RequestContext) {
  return streamText({
    model: models('router'),
    system: ORCHESTRATOR_PROMPT,
    messages: context.messages,
    tools: {
      sql_agent: createAgentTool(sqlAgentConfig),
      business_agent: createAgentTool(businessAgentConfig),
      dashboard_agent: createAgentTool(dashboardAgentConfig),
      comparison_agent: createAgentTool(comparisonAgentConfig),
      external_agent: createAgentTool(externalAgentConfig),
      export_agent: createAgentTool(exportAgentConfig),
      simulation_agent: createAgentTool(simulationAgentConfig),
    },
    stopWhen: stepCountIs(10),
    experimental_context: {
      dataset: context.dataset,
      filters: context.filters,
      dashboardState: context.dashboardState,
    },
  });
}
app/api/chat/route.ts — Simplificado
export async function POST(req: Request) {
  const { messages, context, dataset, filters } = await req.json();
  const user = await verifyAuth(req);
  await checkDatasetAccess(user, dataset);

  const result = createOrchestrator({
    messages: convertToModelMessages(messages),
    dataset,
    filters,
    dashboardState: context,
  });

  return result.toUIMessageStreamResponse();
}

Separacao de Responsabilidades

Config (shared/config/agents/)
O que: Contexto estatico, prompts, schemas
Quem edita: Product/domain experts
Frequencia: Quando regras de negocio mudam
Features (features/ai-agents/)
O que: Logica de execucao, orquestracao
Quem edita: Devs
Frequencia: Quando adiciona agentes ou tools
Widgets (widgets/ai-sidebar/)
O que: UI rendering, step indicators
Quem edita: Frontend devs
Frequencia: Quando UX muda
